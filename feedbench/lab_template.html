Add your own rule

Rules live in feedbench/feedbench/custom_rules.py. Rebuild this page with OASIS:

from .ranking import register_rule

def my_popularity(p, cfg):              # p.endorsers_by_comm = {community: distinct endorsers}
    return float(sum(min(cfg.community_cap, e) for e in p.endorsers_by_comm.values()))

register_rule("my_rule", "M · My rule", popularity=my_popularity)

# then:  python build_lab.py --rules total_endorsements my_rule
